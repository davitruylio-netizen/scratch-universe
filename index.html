Inicializar o Blockly na tela const workspace = Blockly.inject('blocklyDiv', { toolbox: document.getElementById('toolbox'), scrollbars: true, trashcan: true }); // Contexto do Canvas para o jogo const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); let player = { x: 180, y: 180, size: 40, color: '#ff0000' }; function drawPlayer() { ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.fillStyle = player.color; ctx.fillRect(player.x, player.y, player.size, player.size); } drawPlayer(); // 2. Criar a nossa "Linguagem de Programação em Blocos" // Bloco: Quando Começar Blockly.Blocks['when_start'] = { init: function() { this.appendDummyInput().appendField("🏁 Quando Clicar em Rodar"); this.setNextStatement(true, null); this.setColour(45); this.setTooltip("Ponto de partida do jogo"); } }; javascript.javascriptGenerator.forBlock['when_start'] = function(block) { return '// Inicio do Jogo\n'; }; // Bloco: Mover Jogador Blockly.Blocks['move_player'] = { init: function() { this.appendDummyInput() .appendField("Mover") .appendField(new Blockly.FieldDropdown([ ["➡️ Direita", "RIGHT"], ["⬅️ Esquerda", "LEFT"], ["⬆️ Cima", "UP"], ["⬇️ Baixo", "DOWN"] ]), "DIR") .appendField("em") .appendField(new Blockly.FieldNumber(20), "STEPS") .appendField("pixels"); this.setPreviousStatement(true, null); this.setNextStatement(true, null); this.setColour(210); } }; javascript.javascriptGenerator.forBlock['move_player'] = function(block, generator) { const dir = block.getFieldValue('DIR'); const steps = block.getFieldValue('STEPS'); return `movePlayer("${dir}", ${steps});\n`; }; // Bloco: Mudar Cor Blockly.Blocks['set_color'] = { init: function() { this.appendDummyInput() .appendField("Mudar cor do personagem para") .appendField(new Blockly.FieldColour("#ff0000"), "COLOR"); this.setPreviousStatement(true, null); this.setNextStatement(true, null); this.setColour(270); } }; javascript.javascriptGenerator.forBlock['set_color'] = function(block) { const color = block.getFieldValue('COLOR'); return `player.color = "${color}"; drawPlayer();\n`; }; // 3. Funções de Suporte do Jogo function movePlayer(direction, steps) { if (direction === 'RIGHT') player.x = Math.min(canvas.width - player.size, player.x + steps); if (direction === 'LEFT') player.x = Math.max(0, player.x - steps); if (direction === 'UP') player.y = Math.max(0, player.y - steps); if (direction === 'DOWN') player.y = Math.min(canvas.height - player.size, player.y + steps); drawPlayer(); } // 4.
